"""One paid controller call for saved ag-022. No answer/judge/retrieval calls."""
import asyncio,hashlib,json,logging,sys
from pathlib import Path
ROOT=Path(__file__).resolve().parents[1]
sys.path[:0]=[str(ROOT/'backend'),str(ROOT/'backend/scripts/eval')]
from dotenv import load_dotenv
from eval_budget import Budget,BudgetStop,install_transport_guard,LIMITS
from benchmark_agentic_rag import Usage,safe_error
from pydantic import ValidationError

OUT=ROOT/'docs/agentic_rag/20261021-ag022-recheck'

def validation_issues(error):
    issues=[];seen=set()
    while error is not None and id(error) not in seen:
        seen.add(id(error))
        if isinstance(error,ValidationError):
            for item in error.errors(include_url=False,include_input=False):
                allowed={'supported_excerpts','missing_core','rewrite','sufficient'}
                location=[v if type(v) is int or v in allowed else 'other' for v in item['loc']]
                kind=item['type'] if item['type'] in {'string_too_long','too_long','missing','bool_parsing','string_type','list_type'} else 'other'
                bounds={k:v for k,v in item.get('ctx',{}).items()
                        if k in {'max_length','min_length','actual_length'} and type(v) is int}
                issues.append({'location':location,'kind':kind,'bounds':bounds})
        error=error.__cause__ or error.__context__
    return issues

async def main():
    if '--allow-paid' not in sys.argv:raise SystemExit('Explicit authorization required')
    OUT.mkdir(exist_ok=True)
    path=OUT/'single-attempt.json'
    if path.exists():raise SystemExit('Already attempted; no automatic retry')
    load_dotenv(ROOT/'backend/.env',override=True)
    from app.rag import agentic
    from app.rag.assessment_diagnostics import AssessmentOutputError
    from app.core import llm
    from app.core.config import get_settings
    from langchain_core.documents import Document
    from google.genai._api_client import BaseApiClient
    get_settings().model_max_retries=0
    ledger=Budget(ROOT/'tmp/eval-budget.sqlite')
    source=ROOT/'docs/agentic_rag/20261021-batch-live-validation/matched100'
    case=next(c for c in json.loads((source/'cases.json').read_text(encoding='utf-8')) if c['attempt_id']=='ag-022')
    previous=next(r for r in (json.loads(s) for s in (source/'attempts.jsonl').read_text(encoding='utf-8').splitlines())
                  if r['attempt_id']=='ag-022' and r['condition']=='agentic')
    docs=[Document(page_content=d['text'],metadata=d['metadata']) for d in previous['documents']][:4]
    context,docs=agentic._scoped_context(docs)
    record={'attempt_id':'ag-022','status':'entered','budget_before':ledger.summary(),
        'stage_caps':LIMITS,'maximum_paid_calls':1,'retrieval_calls':0,'answer_calls':0,'judge_calls':0,
        'stored_context_sha256':hashlib.sha256(previous['context'].encode()).hexdigest(),
        'assessed_context_sha256':hashlib.sha256(context.encode()).hexdigest(),
        'controller_sha256':agentic.PROMPT_SHA256,'diagnostics':[],'validation_issues':[]}
    with ledger.connect() as db:old_rows=list(db.execute('SELECT id,stage,amount,settled FROM costs ORDER BY id'))
    def save():
        record['budget_after']=ledger.summary()
        with ledger.connect() as db:
            now={r[0]:list(r) for r in db.execute('SELECT id,stage,amount,settled FROM costs')}
        record['old_cost_rows_unchanged']=all(now.get(row[0])==list(row) for row in old_rows)
        record['new_cost_rows']=[v for key,v in now.items() if key not in {row[0] for row in old_rows}]
        path.write_text(json.dumps(record,ensure_ascii=False,indent=2),encoding='utf-8')
    save()
    class Capture(logging.Handler):
        def emit(self,event):
            value=getattr(event,'assessment_diagnostic',None)
            if value is not None:record['diagnostics'].append(value)
    log=logging.getLogger('app.rag.assessment_diagnostics');log.setLevel(logging.INFO);log.addHandler(Capture())
    original_factory=llm.get_chat_model
    class Model:
        def __init__(self,inner):self.inner=inner
        def with_structured_output(self,*args,**kwargs):
            chain=self.inner.with_structured_output(*args,**kwargs)
            class Chain:
                async def ainvoke(self,*a,**kw):
                    raw=await chain.ainvoke(*a,**kw)
                    if isinstance(raw,dict):record['validation_issues']=validation_issues(raw.get('parsing_error'))
                    return raw
            return Chain()
    llm.get_chat_model=lambda **kwargs:Model(original_factory(**kwargs))
    install_transport_guard(ledger,'comparison')
    sync,asynchronous=BaseApiClient.request,BaseApiClient.async_request
    count=0
    def check(method):
        nonlocal count
        if method.lower()!='get':
            if count>=1:raise BudgetStop('Single-call diagnostic limit')
            count+=1
    def guarded(client,method,*args,**kwargs):
        check(method);return sync(client,method,*args,**kwargs)
    async def guarded_async(client,method,*args,**kwargs):
        check(method);return await asynchronous(client,method,*args,**kwargs)
    BaseApiClient.request=guarded;BaseApiClient.async_request=guarded_async
    usage=Usage()
    try:
        decision=await agentic.model_assessment(case['question'],{'context':context},config={'callbacks':[usage]})
        valid = decision.sufficient and not decision.missing_core and bool(decision.supported_excerpts) and all(any(e in d.page_content for d in docs) for e in decision.supported_excerpts)
        record.update(status='controller_pass' if valid else 'controller_contract_failed',contract_pass=valid,decision=decision.model_dump())
    except AssessmentOutputError as exc:
        record.update(status='output_validation_failed',failure=exc.diagnostic)
    except BudgetStop as exc:record.update(status='budget_stopped',reason=str(exc))
    except Exception as exc:record.update(status='failed',error=safe_error(exc))
    finally:
        record.update(usage=usage.calls,transport_attempts=count)
        save()
        print(json.dumps({k:record[k] for k in ('status','transport_attempts','budget_after','validation_issues')}))

asyncio.run(main())
